VS Code - Konfigurasi Variabel Lingkungan API (Soal 2.c)
📄 frontend/.env & frontend/.env.example
# Alamat Backend Laravel diatur lewat Environment Variable:
VITE_API_URL=http://localhost:8000
📄 frontend/src/api/axios.js
import axios from 'axios'

// 1. Alamat API TIDAK ditulis mati (hardcoded), melainkan membaca konfigurasi env:
const apiUrl = import.meta.env.VITE_API_URL || 'http://localhost:8000'

const apiClient = axios.create({
  baseURL: apiUrl,
  headers: { 'Accept': 'application/json', 'Content-Type': 'application/json' }
})

// 2. Interceptor menyisipkan token Sanctum ke Authorization Bearer di setiap request:
apiClient.interceptors.request.use((config) => {
  const token = localStorage.getItem('token')
  if (token) {
    config.headers.Authorization = `Bearer ${token}`
  }
  return config
})

export default apiClient
